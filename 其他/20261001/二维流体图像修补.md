<script type="module">
/* =========================================================================
   二维 Navier–Stokes 图像修补  (Bertalmío / Bertozzi / Sapiro, CVPR 2001)
   -------------------------------------------------------------------------
   设图像灰度 L 为流函数，速度场 v = ∇⊥L = (-L_y, L_x)，涡量 ω = ΔL。
      ① 涡量输运：  ω_t + v·∇ω = ν∇²ω        (半拉格朗日回溯 + 粘性扩散)
      ② 泊松重构：  ΔL = ω  在 Ω 内，∂Ω 上 Dirichlet 条件 = 已知像素
   已知像素全程作为边界条件被冻结，信息沿着等值线（流线）被“吹”进空洞。
   ========================================================================= */

/* ------------------------------- WGSL ---------------------------------- */
const WGSL = /* wgsl */`
struct Params {
  wh : vec4<f32>,   // w, h, 1/w, 1/h
  p0 : vec4<f32>,   // flowStep, nu, smooth, _
  p1 : vec4<f32>,   // overlayMode, iter, blendT, _
  p2 : vec4<f32>,   // meanR, meanG, meanB, _
  p3 : vec4<f32>,   // rectX0, rectY0, rectX1, rectY1
};

@group(0) @binding(0) var<uniform> U : Params;
@group(0) @binding(1) var<storage, read>       imgIn    : array<vec4<f32>>;
@group(0) @binding(2) var<storage, read_write> imgOut   : array<vec4<f32>>;
@group(0) @binding(3) var<storage, read>       omegaIn  : array<vec4<f32>>;
@group(0) @binding(4) var<storage, read_write> omegaOut : array<vec4<f32>>;
@group(0) @binding(5) var<storage, read>       maskBuf  : array<f32>;
@group(0) @binding(6) var<storage, read>       origBuf  : array<vec4<f32>>;
@group(0) @binding(7) var                        outTex : texture_storage_2d<rgba8unorm, write>;
@group(0) @binding(8) var<storage, read>       selBuf   : array<f32>;

fn W() -> i32 { return i32(U.wh.x); }
fn H() -> i32 { return i32(U.wh.y); }

fn IDX(x : i32, y : i32) -> u32 {
  let cx = clamp(x, 0, W() - 1);
  let cy = clamp(y, 0, H() - 1);
  return u32(cy) * u32(W()) + u32(cx);
}
fn getImg(x : i32, y : i32) -> vec4<f32> { return imgIn[IDX(x, y)]; }
fn lum(c : vec3<f32>) -> f32 { return dot(c, vec3<f32>(0.299, 0.587, 0.114)); }

// 双线性采样涡量场（半拉格朗日回溯用）
fn sampleOmega(fx : f32, fy : f32) -> vec4<f32> {
  let x = clamp(fx, 0.0, f32(W() - 1));
  let y = clamp(fy, 0.0, f32(H() - 1));
  let x0 = i32(floor(x));
  let y0 = i32(floor(y));
  let x1 = min(x0 + 1, W() - 1);
  let y1 = min(y0 + 1, H() - 1);
  let tx = x - f32(x0);
  let ty = y - f32(y0);
  let w = u32(W());
  let a = omegaIn[u32(y0) * w + u32(x0)];
  let b = omegaIn[u32(y0) * w + u32(x1)];
  let c = omegaIn[u32(y1) * w + u32(x0)];
  let d = omegaIn[u32(y1) * w + u32(x1)];
  return mix(mix(a, b, tx), mix(c, d, tx), ty);
}

fn hsv2rgb(c : vec3<f32>) -> vec3<f32> {
  let K = vec3<f32>(1.0, 2.0 / 3.0, 1.0 / 3.0);
  let p = abs(fract(c.xxx + K) * 6.0 - 3.0);
  return c.z * mix(vec3<f32>(1.0), clamp(p - 1.0, vec3<f32>(0.0), vec3<f32>(1.0)), c.y);
}

/* ---------- 0. 初始化：已知区保留原图，空洞填均值 ---------- */
@compute @workgroup_size(8, 8, 1)
fn csReset(@builtin(global_invocation_id) gid : vec3<u32>) {
  let x = i32(gid.x); let y = i32(gid.y);
  if (x >= W() || y >= H()) { return; }
  let i = u32(y) * u32(W()) + u32(x);
  if (maskBuf[i] > 0.5) {
    imgOut[i] = origBuf[i];
  } else {
    imgOut[i] = vec4<f32>(U.p2.rgb, 1.0);
  }
}

/* ---------- 1. 涡量：ω = ΔL  （逐通道） ---------- */
@compute @workgroup_size(8, 8, 1)
fn csVorticity(@builtin(global_invocation_id) gid : vec3<u32>) {
  let x = i32(gid.x); let y = i32(gid.y);
  if (x >= W() || y >= H()) { return; }
  let i = u32(y) * u32(W()) + u32(x);
  let c = getImg(x, y).rgb;
  let s = getImg(x + 1, y).rgb + getImg(x - 1, y).rgb
        + getImg(x, y + 1).rgb + getImg(x, y - 1).rgb;
  omegaOut[i] = vec4<f32>(s - 4.0 * c, 0.0);
}

/* ---------- 2. 涡量输运：半拉格朗日回溯 + ν∇²ω + Dirichlet ---------- */
@compute @workgroup_size(8, 8, 1)
fn csAdvect(@builtin(global_invocation_id) gid : vec3<u32>) {
  let x = i32(gid.x); let y = i32(gid.y);
  if (x >= W() || y >= H()) { return; }
  let i = u32(y) * u32(W()) + u32(x);

  // v = ∇⊥L  （等值线方向 → 流线）
  let lx = lum(getImg(x + 1, y).rgb) - lum(getImg(x - 1, y).rgb);
  let ly = lum(getImg(x, y + 1).rgb) - lum(getImg(x, y - 1).rgb);
  var v = vec2<f32>(-ly, lx);
  let s = length(v);
  let step = U.p0.x;
  if (s > 1e-8) { v = v / s * step; } else { v = vec2<f32>(0.0, 0.0); }

  // 回溯采样
  let p = vec2<f32>(f32(x), f32(y)) - v;
  var w = sampleOmega(p.x, p.y);

  // 粘性扩散项 ν∇²ω
  let nu = U.p0.y;
  if (nu > 0.0) {
    let lw = omegaIn[IDX(x + 1, y)] + omegaIn[IDX(x - 1, y)]
           + omegaIn[IDX(x, y + 1)] + omegaIn[IDX(x, y - 1)] - 4.0 * omegaIn[i];
    w = w + nu * lw;
  }

  // 已知区域：涡量被冻结（Dirichlet 边界条件）
  if (maskBuf[i] > 0.5) { w = omegaIn[i]; }
  omegaOut[i] = w;
}

/* ---------- 3. 泊松求解：ΔL = ω（Jacobi，仅更新空洞） ---------- */
@compute @workgroup_size(8, 8, 1)
fn csJacobi(@builtin(global_invocation_id) gid : vec3<u32>) {
  let x = i32(gid.x); let y = i32(gid.y);
  if (x >= W() || y >= H()) { return; }
  let i = u32(y) * u32(W()) + u32(x);
  let c = getImg(x, y);
  if (maskBuf[i] > 0.5) { imgOut[i] = c; return; }     // 冻结已知像素
  let s = getImg(x + 1, y) + getImg(x - 1, y)
        + getImg(x, y + 1) + getImg(x, y - 1);
  let w = omegaIn[i].rgb;
  imgOut[i] = vec4<f32>((s.rgb - w) * 0.25, 1.0);
}

/* ---------- 4. 空洞内各向同性平滑（抑制数值噪声） ---------- */
@compute @workgroup_size(8, 8, 1)
fn csSmooth(@builtin(global_invocation_id) gid : vec3<u32>) {
  let x = i32(gid.x); let y = i32(gid.y);
  if (x >= W() || y >= H()) { return; }
  let i = u32(y) * u32(W()) + u32(x);
  let c = getImg(x, y);
  if (maskBuf[i] > 0.5) { imgOut[i] = c; return; }
  let s = getImg(x + 1, y) + getImg(x - 1, y)
        + getImg(x, y + 1) + getImg(x, y - 1);
  let k = U.p0.z;
  imgOut[i] = vec4<f32>(c.rgb + k * (s.rgb - 4.0 * c.rgb), 1.0);
}

/* ---------- 5. 画笔：在脏矩形内按 sel 指令重置像素 ---------- */
@compute @workgroup_size(8, 8, 1)
fn csPaint(@builtin(global_invocation_id) gid : vec3<u32>) {
  let x = i32(gid.x) + i32(U.p3.x);
  let y = i32(gid.y) + i32(U.p3.y);
  if (f32(x) >= U.p3.z || f32(y) >= U.p3.w) { return; }
  if (x >= W() || y >= H()) { return; }
  let i = u32(y) * u32(W()) + u32(x);
  let s = selBuf[i];
  if (s < 0.5) {              // 0 = 保持不变
    imgOut[i] = imgIn[i];
  } else if (s < 1.5) {       // 1 = 还原原图（擦除）
    imgOut[i] = origBuf[i];
  } else {                    // 2 = 重置为均值（新挖的洞）
    imgOut[i] = vec4<f32>(U.p2.rgb, 1.0);
  }
}

/* ---------- 6. 显示：合成到 rgba8unorm 存储纹理 ---------- */
@compute @workgroup_size(8, 8, 1)
fn csDisplay(@builtin(global_invocation_id) gid : vec3<u32>) {
  let x = i32(gid.x); let y = i32(gid.y);
  if (x >= W() || y >= H()) { return; }
  let i = u32(y) * u32(W()) + u32(x);

  let res = imgIn[i].rgb;
  let org = origBuf[i].rgb;
  let m   = maskBuf[i];
  let t   = clamp(U.p1.z, 0.0, 1.0);
  var col = mix(org, res, t);

  let mode = i32(U.p1.x + 0.5);

  if (mode == 1) {
    // 掩码叠加：空洞部分打上棋盘色
    let hole = 1.0 - m;
    let ck = select(0.82, 1.0, ((x / 8 + y / 8) % 2) == 0);
    col = mix(col, vec3<f32>(0.16, 0.72, 1.0) * ck, hole * 0.55);
  } else if (mode == 2) {
    // 流速场可视化：色相 = 方向，亮度 = 梯度强度
    let lx = lum(getImg(x + 1, y).rgb) - lum(getImg(x - 1, y).rgb);
    let ly = lum(getImg(x, y + 1).rgb) - lum(getImg(x, y - 1).rgb);
    let v  = vec2<f32>(-ly, lx);
    let mag = clamp(length(v) * 28.0, 0.0, 1.0);
    let h   = (atan2(v.y, v.x) + 3.14159265) / 6.28318530;
    let cc  = hsv2rgb(vec3<f32>(h, 0.9, 0.12 + 0.88 * mag));
    col = mix(col * 0.4, cc, 0.92);
  }

  textureStore(outTex, vec2<i32>(x, y), vec4<f32>(clamp(col, vec3<f32>(0.0), vec3<f32>(1.0)), 1.0));
}
`;

const WGSL_RENDER = /* wgsl */`
struct VOut {
  @builtin(position) pos : vec4<f32>,
  @location(0) uv : vec2<f32>,
};
@vertex
fn vs(@builtin(vertex_index) vi : u32) -> VOut {
  var pts = array<vec2<f32>, 3>(
    vec2<f32>(-1.0, -1.0),
    vec2<f32>( 3.0, -1.0),
    vec2<f32>(-1.0,  3.0)
  );
  var o : VOut;
  let p = pts[vi];
  o.pos = vec4<f32>(p, 0.0, 1.0);
  o.uv  = vec2<f32>((p.x + 1.0) * 0.5, 1.0 - (p.y + 1.0) * 0.5);
  return o;
}
@group(0) @binding(0) var samp : sampler;
@group(0) @binding(1) var tex  : texture_2d<f32>;
@fragment
fn fs(in : VOut) -> @location(0) vec4<f32> {
  return textureSample(tex, samp, in.uv);
}

function loadFromSource(src, sw, sh) {
  const scale = Math.min(1, MAXDIM / Math.max(sw, sh));
  const w = Math.max(16, Math.round(sw * scale));
  const h = Math.max(16, Math.round(sh * scale));

  const c = document.createElement('canvas');
  c.width = w; c.height = h;
  const cx = c.getContext('2d', { willReadFrequently: true });
  cx.drawImage(src, 0, 0, w, h);
  const data = cx.getImageData(0, 0, w, h).data;

  alloc(w, h);

  origData  = new Float32Array(N * 4);
  uploadBuf = new Float32Array(N * 4);
  maskData  = new Float32Array(N).fill(1);
  selData   = new Float32Array(N);

  let mr = 0, mg = 0, mb = 0;
  for (let i = 0; i < N; i++) {
    const k = i * 4;
    const r = data[k] / 255, g = data[k + 1] / 255, b = data[k + 2] / 255;
    origData[k] = r; origData[k + 1] = g; origData[k + 2] = b; origData[k + 3] = 1;
    mr += r; mg += g; mb += b;
  }
  meanColor = [mr / N, mg / N, mb / N];

  device.queue.writeBuffer(origBuf, 0, origData);
  device.queue.writeBuffer(maskBuf, 0, maskData);

  canvas.width = w; canvas.height = h;
  ctx.configure({ device, format: presentFormat, alphaMode: 'opaque' });

  document.getElementById('empty').style.display = 'none';
  document.getElementById('sRes').textContent = `${w} × ${h}`;
  resetImage();
}

/* ------------------------------ 重置求解 ------------------------------- */
function resetImage() {
  if (!N) return;
  for (let i = 0; i < N; i++) {
    const k = i * 4;
    if (maskData[i] > 0.5) {
      uploadBuf[k] = origData[k]; uploadBuf[k + 1] = origData[k + 1]; uploadBuf[k + 2] = origData[k + 2];
    } else {
      uploadBuf[k] = meanColor[0]; uploadBuf[k + 1] = meanColor[1]; uploadBuf[k + 2] = meanColor[2];
    }
    uploadBuf[k + 3] = 1;
  }
  device.queue.writeBuffer(img[0], 0, uploadBuf);
  device.queue.writeBuffer(img[1], 0, uploadBuf);
  ci = 0; co = 0; iter = 0;

  readParams();
  device.queue.writeBuffer(uniBuf, 0, params);

  // 预解：先把空洞谐波填充（ΔL = 0, Dirichlet），得到无结构的初始猜测
  const enc = device.createCommandEncoder();
  const cp = enc.beginComputePass();
  cp.setPipeline(PP.vort);
  cp.setBindGroup(0, bgVort(ci, co));
  cp.dispatchWorkgroups(wgX, wgY);
  co ^= 1;
  for (let k = 0; k < 240; k++) {
    cp.setPipeline(PP.jacobi);
    cp.setBindGroup(0, bgJacobi(ci, co));
    cp.dispatchWorkgroups(wgX, wgY);
    ci ^= 1;
  }
  cp.end();
  device.queue.submit([enc.finish()]);
}

/* --------------------------- 一次 Navier–Stokes 外迭代 ------------------ */
function runIteration(cp) {
  // ① ω = ΔL
  cp.setPipeline(PP.vort);
  cp.setBindGroup(0, bgVort(ci, co));
  cp.dispatchWorkgroups(wgX, wgY);
  co ^= 1;

  // ② 涡量输运 ω_t + v·∇ω = ν∇²ω
  cp.setPipeline(PP.advect);
  cp.setBindGroup(0, bgAdvect(ci, co));
  cp.dispatchWorkgroups(wgX, wgY);
  co ^= 1;

  // ③ 泊松重构 ΔL = ω
  const K = parseInt(ui.kiter.value, 10);
  for (let k = 0; k < K; k++) {
    cp.setPipeline(PP.jacobi);
    cp.setBindGroup(0, bgJacobi(ci, co));
    cp.dispatchWorkgroups(wgX, wgY);
    ci ^= 1;
  }

  // ④ 空洞内轻度平滑
  if (parseFloat(ui.smooth.value) > 0) {
    cp.setPipeline(PP.smooth);
    cp.setBindGroup(0, bgSmooth(ci));
    cp.dispatchWorkgroups(wgX, wgY);
    ci ^= 1;
  }
}

/* ------------------------------ 画笔交互 ------------------------------- */
let painting = false, lastPt = null, paintHole = true;
let dirty = false;
const curRect = { x0: 1e9, y0: 1e9, x1: -1, y1: -1 };

function growRect(x0, y0, x1, y1) {
  curRect.x0 = Math.min(curRect.x0, x0);
  curRect.y0 = Math.min(curRect.y0, y0);
  curRect.x1 = Math.max(curRect.x1, x1);
  curRect.y1 = Math.max(curRect.y1, y1);
}

function stamp(cx, cy, r) {
  const x0 = Math.max(0, Math.floor(cx - r)), x1 = Math.min(W - 1, Math.ceil(cx + r));
  const y0 = Math.max(0, Math.floor(cy - r)), y1 = Math.min(H - 1, Math.ceil(cy + r));
  const r2 = r * r;
  const nv = paintHole ? 0 : 1;
  for (let y = y0; y <= y1; y++) {
    const row = y * W;
    const dy = y - cy;
    for (let x = x0; x <= x1; x++) {
      const dx = x - cx;
      if (dx * dx + dy * dy > r2) continue;
      const i = row + x;
      if (maskData[i] === nv) continue;
      maskData[i] = nv;
      selData[i] = paintHole ? 2 : 1;   // 2=重置为均值, 1=还原原图
    }
  }
  if (x1 >= x0 && y1 >= y0) growRect(x0, y0, x1, y1);
}

function strokeTo(x, y) {
  const r = parseFloat(ui.brushR.value);
  if (lastPt) {
    const dx = x - lastPt.x, dy = y - lastPt.y;
    const dist = Math.hypot(dx, dy);
    const steps = Math.max(1, Math.ceil(dist / Math.max(1, r * 0.35)));
    for (let s = 1; s <= steps; s++) {
      stamp(lastPt.x + dx * s / steps, lastPt.y + dy * s / steps, r);
    }
  } else {
    stamp(x, y, r);
  }
  lastPt = { x, y };
  dirty = true;
}

function toImageCoords(e) {
  const r = canvas.getBoundingClientRect();
  return {
    x: (e.clientX - r.left) * (canvas.width / r.width),
    y: (e.clientY - r.top) * (canvas.height / r.height),
  };
}

canvas.addEventListener('pointerdown', (e) => {
  if (!N) return;
  painting = true;
  lastPt = null;
  canvas.setPointerCapture(e.pointerId);
  const p = toImageCoords(e);
  strokeTo(p.x, p.y);
});

/* 每个渲染帧把画笔修改刷到 GPU */
function flushPaint() {
  if (!dirty || !N) return;
  const x0 = Math.max(0, curRect.x0 | 0);
  const y0 = Math.max(0, curRect.y0 | 0);
  const x1 = Math.min(W - 1, curRect.x1 | 0);
  const y1 = Math.min(H - 1, curRect.y1 | 0);

  if (x1 >= x0 && y1 >= y0) {
    const rows = y1 - y0 + 1;
    if (rows > H * 0.6) {
      device.queue.writeBuffer(maskBuf, 0, maskData);
      device.queue.writeBuffer(selBuf, 0, selData);
    } else {
      for (let y = y0; y <= y1; y++) {
        device.queue.writeBuffer(maskBuf, y * W * 4, maskData, y * W, W);
        device.queue.writeBuffer(selBuf,  y * W * 4, selData,  y * W, W);
      }
    }

    // 脏矩形写入 uniform（仅 csPaint 使用）
    readParams();
    params[16] = x0; params[17] = y0; params[18] = x1 + 1; params[19] = y1 + 1;
    device.queue.writeBuffer(uniBuf, 0, params);

    const dx = Math.ceil((x1 - x0 + 1) / WG);
    const dy = Math.ceil((y1 - y0 + 1) / WG);
    const enc = device.createCommandEncoder();
    const cp = enc.beginComputePass();
    cp.setPipeline(PP.paint);
    cp.setBindGroup(0, bgPaint(ci));
    cp.dispatchWorkgroups(dx, dy);
    cp.setBindGroup(0, bgPaint(1 - ci));
    cp.dispatchWorkgroups(dx, dy);
    cp.end();
    device.queue.submit([enc.finish()]);

    // 清理 selData 脏区
    for (let y = y0; y <= y1; y++) selData.fill(0, y * W + x0, y * W + x1 + 1);
  }

  curRect.x0 = 1e9; curRect.y0 = 1e9; curRect.x1 = -1; curRect.y1 = -1;
  if (iter > 0 && running === false) { /* 保持当前迭代数 */ }
  dirty = false;
}

/* ------------------------------ 主渲染循环 ----------------------------- */
let lastMs = performance.now(), accIter = 0, accT = performance.now(), ips = 0, frameMs = 0;

function tick(now) {
  if (!N) { requestAnimationFrame(tick); return; }
  const t0 = performance.now();

  flushPaint();

  readParams();
  device.queue.writeBuffer(uniBuf, 0, params);

  if (running && iter < maxIter) {
    const enc = device.createCommandEncoder();
    const cp = enc.beginComputePass();
    const spf = parseInt(ui.spf.value, 10);
    let s = 0;
    while (s < spf && iter < maxIter) { runIteration(cp); iter++; accIter++; s++; }
    cp.end();
    device.queue.submit([enc.finish()]);
    if (iter >= maxIter) { running = false; syncRunBtn(); }
  }

  // 显示
  const enc2 = device.createCommandEncoder();
  const cp2 = enc2.beginComputePass();
  cp2.setPipeline(PP.display);
  cp2.setBindGroup(0, bgDisplay(ci));
  cp2.dispatchWorkgroups(wgX, wgY);
  cp2.end();

  const rp = enc2.beginRenderPass({
    colorAttachments: [{
      view: ctx.getCurrentTexture().createView(),
      loadOp: 'clear', storeOp: 'store',
      clearValue: { r: 0.02, g: 0.03, b: 0.04, a: 1 },
    }],
  });
  rp.setPipeline(renderPipeline);
  rp.setBindGroup(0, renderBG);
  rp.draw(3);
  rp.end();
  device.queue.submit([enc2.finish()]);

  frameMs = frameMs * 0.85 + (performance.now() - t0) * 0.15;

  const dt = now - accT;
  if (dt > 400) {
    ips = accIter / (dt / 1000);
    accIter = 0; accT = now;
    document.getElementById('sIter').textContent = iter;
    document.getElementById('sIps').textContent = ips.toFixed(1) + ' 步/秒';
    document.getElementById('sMs').textContent = frameMs.toFixed(2) + ' ms/帧';
    document.getElementById('hud').textContent = `迭代 ${iter} · ${ips.toFixed(0)} 步/秒 · ${W}×${H}`;
  }
  requestAnimationFrame(tick);
}

/* 程序化示例图（强等值线，便于观察修补效果） */
function makeSample(w = 768, h = 512) {
  const c = document.createElement('canvas');
  c.width = w; c.height = h;
  const x = c.getContext('2d');

  const g = x.createLinearGradient(0, 0, w, h);
  g.addColorStop(0.00, '#0d2438');
  g.addColorStop(0.45, '#1b4a72');
  g.addColorStop(0.80, '#7fa7c6');
  g.addColorStop(1.00, '#e6eef5');
  x.fillStyle = g; x.fillRect(0, 0, w, h);

  // 同心环
  for (let i = 0; i < 30; i++) {
    x.beginPath();
    x.arc(w * 0.30, h * 0.52, 10 + i * 8.6, 0, Math.PI * 2);
    x.strokeStyle = `hsl(${(i * 11 + 190) % 360} 72% ${30 + (i % 8) * 6}%)`;
    x.lineWidth = 2.4; x.stroke();
  }
  // 螺旋
  x.beginPath();
  for (let t = 0; t < 60; t += 0.06) {
    const r = 14 + t * 4.2;
    const px = w * 0.72 + Math.cos(t) * r;
    const py = h * 0.42 + Math.sin(t) * r * 0.78;
    t === 0 ? x.moveTo(px, py) : x.lineTo(px, py);
  }
  x.strokeStyle = '#ffd166'; x.lineWidth = 5; x.lineJoin = 'round'; x.stroke();

  // 波浪
  for (let k = 0; k < 3; k++) {

### 修补机制与交互流程

页面把图像当作流函数，用GPU迭代求解，让已知像素像边界条件一样把信息“吹”进空洞。

- **上传与标注**：您可通过“上传图片”或“示例图”加载图像，然后用笔刷在画布上涂抹来圈定空洞区域，支持调整笔刷半径和涂抹/擦除模式。
- **求解控制**：点击“开始迭代”后，算法会在每个渲染帧执行多轮Navier–Stokes迭代，包含涡量计算、半拉格朗日输运、泊松重构和平滑处理；拖动滑杆可调整每帧外迭代次数、泊松迭代数、最大迭代数以及流速、涡量扩散ν和平滑σ。
- **实时反馈**：右侧面板实时显示分辨率、已迭代次数、迭代速度和GPU耗时；显示区提供“掩码区域”和“流速场∇⊥I”叠加视图，还能通过“对比原图”滑杆查看修补前后差异。
- **结果导出**：点击“导出PNG”可将当前修补结果保存为图片文件。

   WebGPU · 二维 Navier–Stokes 图像修补
   算法（Bertalmio et al. 的流体重构思路）：
     1) 由图像亮度梯度得到等照度线方向  V = ∇⊥I
     2) 把图像 Laplacian（"涡量" ω = ΔI）沿 V 平流并扩散

  res: 512, dt: 1.0, speed: 1.0, nu: 0.05, eps: 0.02,
  relax: 1.4, relaxIter: 4, steps: 2, prefill: 160,
  brush: 30, erase: false, showMask: true, useNS: true, running: true,
};

/* ---------------- WGSL ---------------- */
const FMT_F = 'rgba32float';

const COMMON = /* wgsl */`
struct Params {
  size   : vec2f,
  dt     : f32,
  nu     : f32,
  speed  : f32,
  relax  : f32,
  eps    : f32,
  parity : f32,
  omega  : f32,
  pad0   : f32,
  pad1   : f32,
  pad2   : f32,
};
`;

/* ---- pass 1 : 计算速度场 V 与涡量 ω ---- */
const WGSL_PREPARE = COMMON + /* wgsl */`
@group(0) @binding(0) var<uniform> P : Params;
@group(0) @binding(1) var srcImg : texture_2d<f32>;
@group(0) @binding(2) var dstVel : texture_storage_2d<rgba32float, write>;
@group(0) @binding(3) var dstW   : texture_storage_2d<rgba32float, write>;

fn ld(x : i32, y : i32) -> vec4f {
  let W = i32(P.size.x); let H = i32(P.size.y);
  return textureLoad(srcImg, vec2i(clamp(x, 0, W - 1), clamp(y, 0, H - 1)), 0);
}
fn lum(c : vec3f) -> f32 { return dot(c, vec3f(0.2126, 0.7152, 0.0722)); }

@compute @workgroup_size(8, 8)
fn main(@builtin(global_invocation_id) gid : vec3u) {
  let W = i32(P.size.x); let H = i32(P.size.y);
  let x = i32(gid.x); let y = i32(gid.y);
  if (x >= W || y >= H) { return; }

  let c  = ld(x, y);
  let cl = ld(x - 1, y);
  let cr = ld(x + 1, y);
  let cu = ld(x, y - 1);
  let cd = ld(x, y + 1);

  // 亮度梯度
  let gx = (lum(cr.rgb) - lum(cl.rgb)) * 0.5;
  let gy = (lum(cd.rgb) - lum(cu.rgb)) * 0.5;
  let gm = sqrt(gx * gx + gy * gy);

  // 等照度线方向  V = ∇⊥I = (-I_y, I_x)，并按 |∇I| 自适应归一化
  let inv = P.speed / (gm + P.eps);
  let vx = -gy * inv;
  let vy =  gx * inv;

  // 涡量 ω = ΔI（逐通道）
  let lap = (cl.rgb + cr.rgb + cu.rgb + cd.rgb) - 4.0 * c.rgb;

  textureStore(dstVel, vec2i(x, y), vec4f(vx, vy, gm, 0.0));
  textureStore(dstW,   vec2i(x, y), vec4f(lap, 0.0));
}
`;

/* ---- pass 2 : 涡量沿等照度线半拉格朗日平流 + 扩散 ---- */
const WGSL_TRANSPORT = COMMON + /* wgsl */`
@group(0) @binding(0) var<uniform> P : Params;
@group(0) @binding(1) var srcW   : texture_2d<f32>;
@group(0) @binding(2) var srcVel : texture_2d<f32>;
@group(0) @binding(3) var dstW   : texture_storage_2d<rgba32float, write>;

fn sampleW(x : f32, y : f32) -> vec3f {
  let Wd = i32(P.size.x); let Hd = i32(P.size.y);
  let fx = clamp(x, 0.0, f32(Wd) - 1.0001);
  let fy = clamp(y, 0.0, f32(Hd) - 1.0001);
  let x0 = i32(floor(fx)); let y0 = i32(floor(fy));
  let tx = fx - f32(x0);   let ty = fy - f32(y0);
  let x1 = min(x0 + 1, Wd - 1);
  let y1 = min(y0 + 1, Hd - 1);
  let c00 = textureLoad(srcW, vec2i(x0, y0), 0).rgb;
  let c10 = textureLoad(srcW, vec2i(x1, y0), 0).rgb;
  let c01 = textureLoad(srcW, vec2i(x0, y1), 0).rgb;
  let c11 = textureLoad(srcW, vec2i(x1, y1), 0).rgb;
  return mix(mix(c00, c10, tx), mix(c01, c11, tx), ty);
}

@compute @workgroup_size(8, 8)
fn main(@builtin(global_invocation_id) gid : vec3u) {
  let Wd = i32(P.size.x); let Hd = i32(P.size.y);
  let x = i32(gid.x); let y = i32(gid.y);
  if (x >= Wd || y >= Hd) { return; }

  let v  = textureLoad(srcVel, vec2i(x, y), 0).xy;
  let px = f32(x) - v.x * P.dt;
  let py = f32(y) - v.y * P.dt;

  let w0  = sampleW(px, py);
  let lap = sampleW(px + 1.0, py) + sampleW(px - 1.0, py)
          + sampleW(px, py + 1.0) + sampleW(px, py - 1.0) - 4.0 * w0;
  let wn  = w0 + P.nu * P.dt * lap;

  textureStore(dstW, vec2i(x, y), vec4f(wn, 0.0));
}
`;

/* ---- pass 3 : 红黑 Gauss–Seidel / SOR 解 Poisson  ΔI = ω ---- */
const WGSL_RELAX = COMMON + /* wgsl */`
@group(0) @binding(0) var<uniform> P : Params;
@group(0) @binding(1) var srcImg  : texture_2d<f32>;
@group(0) @binding(2) var srcW    : texture_2d<f32>;
@group(0) @binding(3) var srcOrig : texture_2d<f32>;
@group(0) @binding(4) var srcMask : texture_2d<f32>;
@group(0) @binding(5) var dstImg  : texture_storage_2d<rgba32float, write>;

@compute @workgroup_size(8, 8)
fn main(@builtin(global_invocation_id) gid : vec3u) {
  let Wd = i32(P.size.x); let Hd = i32(P.size.y);
  let x = i32(gid.x); let y = i32(gid.y);
  if (x >= Wd || y >= Hd) { return; }

  let c = textureLoad(srcImg, vec2i(x, y), 0);

  // 只更新当前颜色的像素，另一色原样写出（红黑 SOR，避免读写冲突）
  if (f32((x + y) & 1) != P.parity) {
    textureStore(dstImg, vec2i(x, y), c);
    return;
  }

  let cl = textureLoad(srcImg, vec2i(clamp(x - 1, 0, Wd - 1), y), 0).rgb;
  let cr = textureLoad(srcImg, vec2i(clamp(x + 1, 0, Wd - 1), y), 0).rgb;
  let cu = textureLoad(srcImg, vec2i(x, clamp(y - 1, 0, Hd - 1)), 0).rgb;
  let cd = textureLoad(srcImg, vec2i(x, clamp(y + 1, 0, Hd - 1)), 0).rgb;

  let om     = textureLoad(srcW, vec2i(x, y), 0).rgb * P.omega;
  let target = (cl + cr + cu + cd - om) * 0.25;   // h = 1 时的 Jacobi 目标
  var nv     = mix(c.rgb, target, P.relax);       // SOR 超松弛

  // Dirichlet 边界：已知像素强制回到原图（软掩码做平滑过渡）
  let k = clamp(textureLoad(srcMask, vec2i(x, y), 0).r, 0.0, 1.0);
  let o = textureLoad(srcOrig, vec2i(x, y), 0).rgb;
  nv = mix(nv, o, k);

  textureStore(dstImg, vec2i(x, y), vec4f(nv, 1.0));
}
`;

/* ---- pass 4 : 笔刷 ---- */
const WGSL_BRUSH = /* wgsl */`
struct Brush {
  p0     : vec2f,
  p1     : vec2f,
  size   : vec2f,
  radius : f32,
  value  : f32,
  pad0   : f32,
  pad1   : f32,
  pad2   : f32,
};
@group(0) @binding(0) var<uniform> B : Brush;
@group(0) @binding(1) var srcMask : texture_2d<f32>;
@group(0) @binding(2) var dstMask : texture_storage_2d<rgba32float, write>;

@compute @workgroup_size(8, 8)
fn main(@builtin(global_invocation_id) gid : vec3u) {
  let Wd = i32(B.size.x); let Hd = i32(B.size.y);
  let xi = i32(gid.x); let yi = i32(gid.y);
  if (xi >= Wd || yi >= Hd) { return; }

  let p  = vec2f(f32(xi) + 0.5, f32(yi) + 0.5);
  let ab = B.p1 - B.p0;
  let t  = clamp(dot(p - B.p0, ab) / max(dot(ab, ab), 1e-6), 0.0, 1.0);
  let q  = B.p0 + ab * t;
  let d  = distance(p, q);
  let a  = 1.0 - smoothstep(B.radius * 0.55, B.radius, d);

  let cur = textureLoad(srcMask, vec2i(xi, yi), 0).r;
  var nv = cur;
  if (a > 0.0) {
    if (B.value < 0.5) { nv = cur * (1.0 - a); }          // 涂抹 → 挖洞
    else               { nv = cur + (1.0 - cur) * a; }    // 擦除 → 恢复
  }
  textureStore(dstMask, vec2i(xi, yi), vec4f(nv, 0.0, 0.0, 1.0));
}
`;

/* ---- pass 5 : 显示 ---- */
const WGSL_RENDER = COMMON + /* wgsl */`
struct RV { @builtin(position) p : vec4f, @location(0) uv : vec2f };
@group(0) @binding(0) var<uniform> P : Params;
@group(0) @binding(1) var imgTex  : texture_2d<f32>;
@group(0) @binding(2) var maskTex : texture_2d<f32>;

@vertex
fn vs(@builtin(vertex_index) i : u32) -> RV {
  var pts = array<vec2f,3>(vec2f(-1.0,-1.0), vec2f(3.0,-1.0), vec2f(-1.0,3.0));
  let p = pts[i];
  var o : RV;
  o.p  = vec4f(p, 0.0, 1.0);
  o.uv = vec2f(p.x * 0.5 + 0.5, 0.5 - p.y * 0.5);
  return o;
}

@fragment
fn fs(in : RV) -> @location(0) vec4f {
  let dim = textureDimensions(imgTex);
  let px  = vec2i(floor(in.uv * vec2f(dim)));
  let c   = vec2i(clamp(px.x, 0, i32(dim.x) - 1), clamp(px.y, 0, i32(dim.y) - 1));

  let col = textureLoad(imgTex, c, 0).rgb;
  if (P.omega > 0.5) {                       // 这里复用 omega 字段作为"显示掩码"开关
    let k = clamp(textureLoad(maskTex, c, 0).r, 0.0, 1.0);
    return vec4f(mix(col, vec3f(0.06, 0.62, 1.0), (1.0 - k) * 0.55), 1.0);
  }
  return vec4f(col, 1.0);
}

function tick(){
  requestAnimationFrame(tick);
  if (!TEX.imgMain) return;

  const enc = device.createCommandEncoder();
  let dispatches = 0;

  if (P.running) {
    const cp = enc.beginComputePass({ label: 'navier-stokes' });

    // —— 预填充：ω=0 的 Laplace 方程 SOR，快速给出光滑初值 ——
    if (pendingPrefill > 0) {
      cp.setPipeline(PIPE.relax);
      for (let k = 0; k < pendingPrefill; k++) {
        cp.setBindGroup(0, BG.relaxApf); cp.dispatchWorkgroups(GW, GH);
        cp.setBindGroup(0, BG.relaxBpf); cp.dispatchWorkgroups(GW, GH);
        dispatches += 2;
      }
      pendingPrefill = 0;
      stepCount = 0;
    }

    // —— 主循环：prepare → transport → relax × N ——
    for (let s = 0; s < P.steps; s++) {
      cp.setPipeline(PIPE.prepare);
      cp.setBindGroup(0, BG.prepare);
      cp.dispatchWorkgroups(GW, GH);

      cp.setPipeline(PIPE.transport);
      cp.setBindGroup(0, BG.transport);
      cp.dispatchWorkgroups(GW, GH);

      cp.setPipeline(PIPE.relax);
      for (let k = 0; k < P.relaxIter; k++) {
        cp.setBindGroup(0, BG.relaxA); cp.dispatchWorkgroups(GW, GH);
        cp.setBindGroup(0, BG.relaxB); cp.dispatchWorkgroups(GW, GH);
      }
      dispatches += 2 + P.relaxIter * 2;
      stepCount++;
    }
    cp.end();
  }

  // 渲染到画布
  const rp = enc.beginRenderPass({
    colorAttachments: [{
      view: ctx.getCurrentTexture().createView(),
      clearValue: { r: 0.03, g: 0.04, b: 0.06, a: 1 },
      loadOp: 'clear', storeOp: 'store',
    }],
  });
  rp.setPipeline(PIPE.render);
  rp.setBindGroup(0, BG.render);
  rp.draw(3);
  rp.end();

  device.queue.submit([enc.finish()]);

  // 统计
  frames++;
  dispAcc += dispatches;
  const now = performance.now();
  if (now - fpsT > 500) {
    fps = frames * 1000 / (now - fpsT);
    frames = 0; fpsT = now;
    sFps.textContent = fps.toFixed(0);
    sStep.textContent = stepCount;
    sDispatch.textContent = dispAcc;
    dispAcc = 0;
  }
}

### 算法流程与交互方式

这个页面把流体力学的思路搬进了图像修补，您可以通过涂抹掩码来指定修补区域，算法会自动把周边结构“推”进空洞。

- **算法思路**：代码基于Bertalmio等人的流体重构方法——先从图像亮度梯度计算等照度线方向，再把图像Laplacian（当作“涡量”）沿该方向平流并扩散，最后在空洞内解Poisson方程，每一步都利用已知像素的Dirichlet边界条件，让外部纹理自然向内延伸。
- **操作与反馈**：上传图片或使用内置的同心圆测试图后，用鼠标在画布上涂抹即可标记需要修补的区域，松开笔刷后自动触发预填充和迭代。右侧面板可随时调整分辨率、时间步、平流速度、粘性以及泊松松弛轮数等参数，并实时观察修补效果。
- **实时统计**：界面会显示当前迭代步数、FPS以及每帧调度次数，方便您判断性能与收敛状态。
---

**优化建议：** 您可以在右侧“求解参数”面板中滑动“分辨率”“平流速度”或“粘性”等滑杆，实时调整修补效果与运算性能；若想替换默认的测试图，直接上传自己的图片并涂抹掩码即可。
